        Integer first = null;

            return -1;

            else if(second ==null ||num > second){
                third =second;
                second = num;

            else if(third == null || num > third){
                third = num;
            }

    }

4)finding missing number in array :
Examples:
Input: arr[] = [1, 2, 3, 5]
Output: 4
Explanation: All the numbers from 1 to 5 are present except 4.
Input: arr[] = [8, 2, 4, 5, 3, 7, 1]
Output: 6
Explanation: All the numbers from 1 to 8 are present except 6.
Input: arr[] = [1]
Output: 2
Explanation: Only 1 is present so the missing element is 2.

class Solution {
    long missingNum(int arr[]) {
        // code here
        long n = arr.length+1;
        long sum = 0;
        for(int i=0;i<arr.length;i++){
            sum = sum+arr[i];
        }
        long formula_sum=(n*(n+1))/2;
        long missing =formula_sum - sum;
        return missing;
    }
}

5)rotate array by 1 time :   (shifting method)
input : [1,2,3,4,5] 
output : [5,1,2,3,4]
T.C : O(n)
S.C : O(1)

class Solution {
    public void rotate(int[] arr) {
        // code here
        int last_element = arr[arr.length-1];
        for(int i=arr.length-1;i>0;i--){
            arr[i]=arr[i-1];
        }
        arr[0]=last_element;
    }
}

import java.util.ArrayList;
public class Solution {
public static void reverse(ArrayList<Integer> arr, int s, int e) {
        while (s < e) {
            int temp = arr.get(s);
            arr.set(s, arr.get(e));
            arr.set(e, temp);
            s++;
            e--;
        }
    }
    public static ArrayList<Integer> rotateArray(ArrayList<Integer> arr, int k) {
        int n = arr.size();
        k = k % n;
        reverse(arr, 0, k - 1);
        reverse(arr, k, n - 1);
        reverse(arr, 0, n - 1);
        return arr;
    }
}

import java.util.HashMap;
class Solution {
    public boolean isSubset(int a[], int b[]) {
        // Your code here
        HashMap <Integer,Integer> map = new HashMap<>();
        //frequency of numbers in array
        for(int num : a){
            map.put(num,map.getOrDefault(num,0)+1);
        }
        for(int subnum :b){
            if(!map.containsKey(subnum) || map.get(subnum) == 0){
                return false;
            }
            map.put(subnum,map.get(subnum)-1);
        }
        return true;
    }
}

10)Remove duplicats elements in an array :                       (striver)
class Solution {
    public int removeDuplicates(int[] nums) {
        int n = nums.length;
        int i=0;
        for(int j=1;j<n;j++){
          if(nums[i]!=nums[j]){
            i++;
            nums[i]=nums[j];
          }
        }
        return i+1;             
    }
}
Array: 1 1 2 2 3 4 4

Initial:
i = 0, j = 1
j = 1
1 == 1 → Duplicate → Do nothing
j = 2
1 != 2 → i++ → nums[i] = 2
Array: 1 2 2 2 3 4 4
j = 3
2 == 2 → Duplicate → Do nothing
j = 4
2 != 3 → i++ → nums[i] = 3
Array: 1 2 3 2 3 4 4
j = 5
3 != 4 → i++ → nums[i] = 4
Array: 1 2 3 4 3 4 4
j = 6
4 == 4 → Duplicate → Do nothing
Return = i + 1 = 4
Unique Elements:
1 2 3 4

Time Complexity
Best Case: O(n)
Average Case: O(n)
Worst Case: O(n)
Reason: j traverses the array only once.
Space Complexity
O(1)
Reason: Only two pointers (i, j) are used; no extra array is created

11)move zeros to end of the array : (Brute force striver)

Input: nums = [0, 1, 4, 0, 5, 2]
Output: [1, 4, 5, 2, 0, 0]
Explanation:  Both the zeroes are moved to the end and the order of the other elements stay the same
class Solution {
    public void moveZeroes(int[] nums) {
        int n = nums.length;
        int zeroCount=0;
        ArrayList <Integer> dup =  new ArrayList<>(); 
        for(int i=0;i<n;i++){
            if(nums[i]!=0){
            dup.add(nums[i]);
            }
            else{
               zeroCount++;
            }
        }
        for(int j=0;j<dup.size();j++){
            nums[j]=dup.get(j);
        }
        int m = dup.size();
        while(zeroCount>0){
            nums[m]=0;
            zeroCount--;
            m++;
        }
    }
}
T.C O(n), s.c O(n)
