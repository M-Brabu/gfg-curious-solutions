class Solution {
    public boolean isAnagram(String s, String t) {
        if(s.length() != t.length()){
            return false;
        }
        HashMap  <Character,Integer> sm =  new HashMap<>();
        HashMap  <Character,Integer> tm =  new HashMap<>();
        for(int i=0;i<s.length();i++){
            sm.put(s.charAt(i),sm.getOrDefault(s.charAt(i),0)+1);
        }
        for(int j=0;j<t.length();j++){
            tm.put(t.charAt(j),tm.getOrDefault(t.charAt(j),0)+1);
        }
        return sm.equals(tm);     
    }
} 
t.c : O(n) , s.c : O(n)

class Solution {
    public boolean isAnagram(String s, String t) {
        if (s.length() != t.length()) {
            return false;
        }
        int[] freq = new int[26];
        for (int i = 0; i < s.length(); i++) {
            freq[s.charAt(i) - 'a']++;
            freq[t.charAt(i) - 'a']--;
        }
        for (int count : freq) {
            if (count != 0) {
                return false;
            }
        }
        return true;
    }
}

class Solution {
    public String largestOddNumber(String num) {
         for(int i =num.length()-1;i>=0;i++){

              return num.substring(0,i+1);

4) palindrome string : e.g : appa - rev - appa palindrome , boy - rev - yob not palindrome
Brute : 
class Solution {
    boolean isPalindrome(String s) {
        StringBuilder sb = new StringBuilder();
        int j=s.length()-1;
        for(int i=0;i<s.length();i++){
            sb.insert(i,s.charAt(j));
            j--;
        }
        String rev = new String(sb);
        if(rev.equals(s)){
            return true;
        }
        return false;
    }
}
t.c : O(n) , s.c : O(n)
optimal : 
class Solution {
    boolean isPalindrome(String s) {
        int left = 0;
        int right = s.length() - 1;

        while (left < right) {
            if (s.charAt(left) != s.charAt(right)) {
                return false;
            }
            left++;
            right--;
        }

        return true;
    }
}
t.c : O(n) , s.c : O(1)

5) Rotate string : e.g : str= "abc"  any possibilty of getting "cab" ? after rotation abc - bca -cab  yes after 2 rotation: "cba" no 
brute :
class Solution {
    public boolean rotateString(String s, String goal) {
      int slen = s.length();
      int goallen = goal.length();
      if(slen != goallen){
        return false;
      }
      for(int i=0;i<slen;i++){         // slen times rotation coz slen possible string generated by s
           s=s.substring(1)+s.charAt(0);
           if(s.equals(goal)){
            return true;
           }
      }
      return false;  
    }
}
t.c : O(n^2) , s.c : O(n)

class Solution {
    public String reverseWords(String s) {
     String [] arr = s.trim().split("\\s+");    -> String s = "the    sky";
String[] arr = s.split(" ");                       Result:"the", "", "", "", "sky"]
                                                   The empty strings ("") are created because split(" ") splits only on a single space.                                                    so use \\s to avoid such in between spaces + one or more spaces  
int start = 0;
    int end = arr.length-1;
     while(start<end){
        String temp = arr[start];
        arr[start] = arr[end] ;
        arr[end] = temp; 
        start++;
        end--;
     }  
     String snew = String.join(" ",arr);
     return snew;
    }
}
t.c : O(n) , s.c : O(n)
