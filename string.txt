class Solution {
    public boolean isAnagram(String s, String t) {
        if(s.length() != t.length()){
            return false;
        }
        HashMap  <Character,Integer> sm =  new HashMap<>();
        HashMap  <Character,Integer> tm =  new HashMap<>();
        for(int i=0;i<s.length();i++){
            sm.put(s.charAt(i),sm.getOrDefault(s.charAt(i),0)+1);
        }
        for(int j=0;j<t.length();j++){
            tm.put(t.charAt(j),tm.getOrDefault(t.charAt(j),0)+1);
        }
        return sm.equals(tm);     
    }
}

class Solution {
    boolean isPalindrome(String s) {
        int left = 0;
        int right = s.length() - 1;

        while (left < right) {
            if (s.charAt(left) != s.charAt(right)) {
                return false;
            }
            left++;
            right--;
        }

        return true;
    }
}

class Solution {
    public boolean rotateString(String s, String goal) {
      int slen = s.length();
      int goallen = goal.length();
      if(slen != goallen){
        return false;
      }
      for(int i=0;i<slen;i++){         // slen times rotation coz slen possible string generated by s
           s=s.substring(1)+s.charAt(0);
           if(s.equals(goal)){
            return true;
           }
      }
      return false;  
    }
}

class Solution {
    public String longestCommonPrefix(String[] strs) {
        Arrays.sort(strs);
        int index = 0;
        String s1 = strs[0];
        String s2 = strs[strs.length-1];
        while(index<s1.length() && index<s2.length()){
            if(s1.charAt(index) == s2.charAt(index)){
                index++;
            }
            else{
                break;
            }
        }
        return index == 0 ? "" : s1.substring(0,index); 
    }
}

10)FiZZ Buzz : i div by 3 & 5 fizzbuzz , i div by 3 fizz , i div by fizz : e.g : n = 5 =>["1","2","Fizz","4","buzz"]
class Solution {
    public List<String> fizzBuzz(int n) {
        ArrayList <String> answer  = new  ArrayList<>();
        for(int i=1;i<=n;i++){
            if(i%3==0 && i%5==0){
               answer.add("FizzBuzz");
            }
            else if(i%3==0){
                answer.add("Fizz");
            }
            else if(i%5==0){
                answer.add("Buzz");
            }
            else{
                String ii = Integer.toString(i);
                answer.add(ii);
            }
        }
        return answer;
    }
}
t.c : O(n) , s.c : O(n)

class Solution {
    public int romanToInt(String s) {
        HashMap <Character,Integer> map =  new HashMap<>();
        map.put('I',1);
        map.put('V',5);
        map.put('X',10);
        map.put('L',50);
        map.put('C',100);
        map.put('D',500);
        map.put('M',1000);
        int result = 0;
        for(int i=0;i<s.length();i++){
            int current = map.get(s.charAt(i));
            if(i<s.length()-1 && current<map.get(s.charAt(i+1))){
             result = result - current;   
            }
            else{
                result = result + current;
            }
        }
        return result;
    }
}
t.c : O(n) ,s.c :O(1)

13)Robot Return to Origin :

Example 1:
Input: moves = "UD"
Output: true
Explanation: The robot moves up once, and then down once. All moves have the same magnitude, so it ended up at the origin where it started. Therefore, we return true.

Example 2:
Input: moves = "LL"
Output: false
Explanation: The robot moves left twice. It ends up two "moves" to the left of the origin. We return false because it is not at the origin at the end of its moves.

class Solution {
    public boolean judgeCircle(String moves) {
        int x=0,y=0;
        int n = moves.length();
        for(int i=0;i<n;i++){
              if(moves.charAt(i)== 'U'){
                 x++;
              }
              else if(moves.charAt(i)== 'D'){
                 x--;
              }
              else if(moves.charAt(i)== 'L'){
                 y++;
              }
              else{
                 y--;
              }
        }
        if(x==0 && y==0){
            return true;
        }
        return false;
    }
}
t.c :O(n), s.c:O(1)
